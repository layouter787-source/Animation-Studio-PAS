cmake_minimum_required(VERSION 3.22.1)
project(AnimationStudioPAS LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(android_native_app_glue STATIC ${ANDROID_NDK}/sources/android/native_app_glue/android_native_app_glue.c)
target_include_directories(android_native_app_glue PUBLIC ${ANDROID_NDK}/sources/android/native_app_glue)
add_library(pas_engine SHARED main.cpp)
target_compile_options(pas_engine PRIVATE -Wall -Wextra -Wpedantic -O2)
target_link_libraries(pas_engine android_native_app_glue android log EGL GLESv3)
